# Components & State

Feature component structure, global components, Zustand stores, and provider composition.

## Feature `_components/` Structure

Each feature has a **container** component and **presenter** components.

```
_components/
├── projects-view.tsx      # container — fetches data, manages interactions
├── project-card.tsx       # presenter — receives props, renders UI
├── project-form.tsx       # form — uses react-hook-form + mutation
└── project-filters.tsx    # presenter — search/filter controls
```

### Container (View)

The main client component. Calls hooks, passes data to presenters.

```tsx
"use client";

import { useProjects } from "../_hooks/use-projects";
import { useProjectsParams } from "../_hooks/use-projects-params";
import { ProjectCard } from "./project-card";

export const ProjectsView = () => {
  const { data } = useProjects();
  const [params, setParams] = useProjectsParams();

  return (
    <div>
      <ProjectFilters params={params} onChange={setParams} />
      <div className="grid gap-4 md:grid-cols-3">
        {data.data.map((project) => (
          <ProjectCard key={project.id} project={project} />
        ))}
      </div>
    </div>
  );
};
```

### Presenter

Receives props, renders UI. No direct hook calls for data fetching.

```tsx
import type { Project } from "../_validations/project-schemas";

interface ProjectCardProps {
  project: Project;
}

export const ProjectCard = ({ project }: ProjectCardProps) => (
  <div className="rounded-lg border p-4">
    <h3>{project.name}</h3>
    <p>{project.description}</p>
  </div>
);
```

### Naming Convention

All files in a feature's `_components/` are prefixed with the feature name:

- `project-card.tsx` (singular for individual items)
- `projects-view.tsx` (plural for collections/containers)
- `project-form.tsx` (singular for forms)

## Zustand Stores

### Ephemeral Store (in `core/stores/`)

For UI state that doesn't need persistence.

```ts
import { create } from "zustand";

interface SidebarStore {
  isOpen: boolean;
  toggle: () => void;
  setOpen: (open: boolean) => void;
}

export const useSidebarStore = create<SidebarStore>()((set) => ({
  isOpen: true,
  toggle: () => set((state) => ({ isOpen: !state.isOpen })),
  setOpen: (isOpen) => set({ isOpen }),
}));
```

### Persistent Store (in `core/storage/`)

For state that survives page reloads (uses Zustand persist middleware).

```ts
import { create } from "zustand";
import { persist } from "zustand/middleware";

interface ThemeStorage {
  theme: "light" | "dark";
  setTheme: (theme: "light" | "dark") => void;
}

export const useThemeStorage = create<ThemeStorage>()(
  persist(
    (set) => ({
      theme: "light",
      setTheme: (theme) => set({ theme }),
    }),
    { name: "theme-storage" }
  )
);
```

## Provider Composition

Compose all providers into a single `root-provider.tsx`, then import in the root layout.

### `core/providers/root-provider.tsx`

```tsx
"use client";

import { QueryClientProvider } from "@tanstack/react-query";
import { NuqsAdapter } from "nuqs/adapters/next/app";
import { getQueryClient } from "@/lib/get-query-client";

export const RootProvider = ({ children }: { children: React.ReactNode }) => {
  const queryClient = getQueryClient();

  return (
    <QueryClientProvider client={queryClient}>
      <NuqsAdapter>{children}</NuqsAdapter>
    </QueryClientProvider>
  );
};
```

### `app/layout.tsx`

```tsx
import { RootProvider } from "@/core/providers/root-provider";

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html>
      <body>
        <RootProvider>{children}</RootProvider>
      </body>
    </html>
  );
}
```
